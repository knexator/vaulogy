    (false . nil) -> false;
    (true . (false . @rest)) -> mostCommonBoolean: @rest;
    (false . (true . @rest)) -> mostCommonBoolean: @rest;
    @other -> ( mostCommonBoolean . helper ): (@other . (nil . nil));
}

    (nil . ( (true . @r1) . (false . @r2))) -> ( mostCommonBoolean . helper ): (nil . (@r1 . @r2));

    ((false . @r) . (@r1 . @r2)) -> ( mostCommonBoolean . helper ): (@r . (@r1 . (false . @r2)));
