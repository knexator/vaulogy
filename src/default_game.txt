// This is a game about programming in a simple language.
// Your goal is to finish all these functions.

// Each function is just a list of hardcoded match cases.
// This one takes a letter from 'a' to 'f' and returns it in uppercase.
uppercase {
    a -> A;
    b -> B;
    c -> C;
    d -> D;
    e -> E;
    f -> F;
}

// As a warm up, finish this function and run the command to see your score.
lowercase {
    A -> a;
    B -> b;
    // C -> c;
    // ...
}

// Words starting with '@' are wildcards and will match anything.
isVowel {
    A -> true;
    E -> true;
    @other -> false;
}

// Values can also be a pair of two values, to any depth.
// There are too many cases to hardcode; use wildcards.
swap {
    ( A . B ) -> ( B . A );
    ( C . D ) -> ( D . C );
    ( A . (C . D) ) -> ( (C . D) . A );
    ( (E . F) . (C . D) ) -> ( (C . D) . (E . F) );
    (((A . a) . (B . b)) . ((C . c) . (D . d))) -> (((C . c) . (D . d)) . ((A . a) . (B . b)));
    ( A . @right ) -> ( @right . A );
    ( @left . A ) -> ( A . @left );
}

// You can call a function on a value before returning it.
firstAsUppercase {
    // hardcoded direct result:
    ( a . @other ) -> A;
    ( b . @other ) -> B;
    // calling the 'uppercase' function instead:
    ( c . @other ) -> uppercase: c;
    ( d . @other ) -> uppercase: d;
}

// You can run other match cases on a value before returning it.
startWithB {
    (@left . @right) -> @left {
        b -> true;
        @else -> false;
    }
}

// Finally, you can combine both: calling a function on a value,
// and then running a match case on the result.
// This function is already solved, as an example.
uppercaseIfVowel {
    @letter -> isVowel: @letter {
        true -> uppercase: @letter;
        false -> @letter;
    }
}

// That's all the rules! You now know the whole language.
// The real game begins now.

firstToUppercase {
    (a . a) -> (A . a);
    (a . b) -> uppercase: a {
        A -> (A . b);
    }
    (@first . c) -> uppercase: @first {
        @FIRST -> (@FIRST . c);
    }
}

pairToUppercase {
    (a . a) -> (A . A);
    (a . b) -> (A . B);
    // you could handcode all 36 cases, for a poor "code size" score.
}

// By the way, you can define your own helper functions.
// They only get counted once for the "code size" score,
// so it's a big win if it's useful for multiple levels.

changeCase {
    a -> A;
    A -> a;
}

letterToBothCases {
    a -> (a . A);
    A -> (a . A);
}

// Hint: functions can call themselves

hasSomeB {
    a -> false;
    b -> true;
    c -> false;
    d -> false;
    e -> false;
    f -> false;

    (a . b) -> true;
    (c . d) -> false;
    ((e . f) . (a . ((c . b) . f))) -> true;
    (((a . c) . (c . a)) . ((c . a) . (a . c))) -> false;

    (@any . b) -> true;
    (b . @any) -> true;
}


// All functions take and return a single value.
// What if we need a list of values instead?
// Here's the convention we will use:
// - We represent empty lists with 'nil'
// - Otherwise, we make a pair where the left half is the first element of the list,
//   and the second half is the rest of the list: (@first . @rest).
// Some examples:
// - '(a . nil)' is a list with one element, 'a' (since the 'rest' is just the empty list)
// - '(b . (a . nil))' is 'b' followed by the previous example; in other words, a list of 2 elements, 'b' and then 'a'.
// - '((a . b) . (c . nil))' is a list of two elements: the first one is '(a . b)', the second one is 'c'.
// This is important! Many of the functions from now on will take or return lists.

// return the second element of a list
// you can assume the input is a list with at least two elements.
second {
    (a . (b . nil)) -> b;
    (a . (b . (c . nil))) -> b;
    (a . (b . (c . (d . nil)))) -> b;
}

listHasSomeB {
    // the empty list has no 'b' elements
    nil -> false;
    (f . nil) -> false;
    (a . (b . nil)) -> true;
    (c . (d . (e . nil))) -> false;

    // The first element of this list is '(b . b)',
    // which is not 'b'!
    ((b . b) . (a . nil)) -> false;
}

uppercaseEachElement {
    nil -> nil;
    (a . nil) -> (A . nil);
    (a . (b . nil)) -> (A . (B . nil));
    (c . (d . (e . nil))) -> (C . (D . (E . nil)));
}

hasAnyVowel{
    nil -> false;
    (a . nil) -> true;
    (b . (c . nil)) -> false;
    (c . (d . (e . nil))) -> false;
}

// reverses a list
reverse {
    nil -> nil;
    (a . nil) -> (a . nil);
    (b . (c . nil)) -> (c . (b . nil));
    (c . (d . (e . nil))) -> (e . (d . (c . nil)));
}

// Given a list of 'true's and 'false's, return the most common.
// Guaranteed to not be a tie.
mostCommonBoolean {
    (true . (false . (true . nil))) -> true;
    (true . (false . (false . (true . (false . nil))))) -> false;
    (true . (true . (true . (false . (false . (false . (false . nil))))))) -> false;
    (true . (false . (true . (false . (true . (false . (true . nil))))))) -> true;
}


// Bonus level: Brainf*ck
// Implement the brainfuck programming language.
// Input is a pair (source_code . stdin), where stdin is a lsit of numbers
// and source_code is a list of symbols:
//  'next': increment the pointer
//  'prev': decrement the pointer
//  'inc': increment the pointed number
//  'dec': decrement the pointed number
//  'in': take a number from stdin and store it at the pointed position
//  'out': store the pointed number in stdout
//  'loop': if the pointed number is zero, skip all code up to the matching 'end'
//  'end': if the pointed number is not zero, go back in code to the matching 'loop'
// You must return the resulting stdout
// Assume the memory tape has no ends, and each cell can get arbitrarily big.
brainfuck {
    // ",,!,!", "1 2 3 4" -> "2 3"
    (
        ( in . ( in . ( out . ( in . ( out . nil))))) . 
        (
            (N . nil) . (
            (N . (N . nil)) . (
            (N . (N . (N . nil))) . (
            (N . (N . (N . (N . nil)))) . 
        nil))))
    ) -> ((N . (N . nil)) . ((N . (N . (N . nil))) . nil));

    // "++!+++!----!" -> "2 5 1"
    (
        ( inc . ( inc . ( out . 
        ( inc . ( inc . (inc . (out . 
        (dec . (dec . (dec . (dec . (out . nil))))))))))))
    . nil) -> ((N . (N . nil)) . ((N . (N . (N . (N . (N . nil))))) . ((N . nil) . nil)));

    // "+>><<!" -> "1"
    (
        ( inc . ( next . ( next . ( prev . ( prev . (out . nil))))))
    . nil) -> ((N . nil) . nil);

    // "+++[->++<]>!" -> "6"
    (
        ( inc . ( inc . ( inc . 
        ( loop . 
        ( dec . (next . (inc . (inc . (prev . 
        ( end . 
        ( next . (out . nil))))))))))))
    . nil) -> ((N . (N . (N . (N . (N . (N . nil)))))) . nil);
}
